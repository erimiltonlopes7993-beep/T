        autoIndex = state.autoIndex or false
        selectedMutations = state.selectedMutations or {Normal = true, Golden = true, Diamond = true, Galaxy = true}

        mutationLabel.Text = mutation
        mutationLabel.TextColor3 = Color3.fromRGB(255,255,255)
        mutationLabel.TextSize = 14
        mutationLabel.Font = Enum.Font.Gotham

local function findPlot()
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return end

    for _, plot in pairs(plots:GetChildren()) do
        local tag = plot:FindFirstChild("BillboardSpawnPart", true)
        if tag then
            local label = tag:FindFirstChild("TextLabel", true)
            if label and label.Text == player.Name then
                return plot
            end
        end
    end
end

local function findMeshPartInModel(model, rarityText)
    local rarityLabel = nil
    
    for _, descendant in pairs(model:GetDescendants()) do
        if descendant.Name == "RarityLabel" and descendant:IsA("TextLabel") then
            if descendant.Text == rarityText then
                rarityLabel = descendant
                break
            end
        end
    end
    
    if rarityLabel then
        for _, descendant in pairs(model:GetDescendants()) do
            if descendant.ClassName == "MeshPart" then
                return descendant
            end
        end
    end
    
    return nil
end

local teleportCFrame = CFrame.new(94.9108829, 70.4400024, 2.796, 1, 0, 0, 0, 1, 0, 0, 0, 1)

local function collectRarity(rarityType)
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
        return
    end
    
    local entitiesFolder = workspace:FindFirstChild("EntitiesFolder")
    if not entitiesFolder then return end
    
    for _, model in pairs(entitiesFolder:GetChildren()) do
        if model:IsA("Model") then
            local meshPart = findMeshPartInModel(model, rarityType)
            if meshPart then
                player.Character.HumanoidRootPart.CFrame = meshPart.CFrame
                task.wait(0.3)
                
                for _, descendant in pairs(model:GetDescendants()) do
                    if descendant:IsA("ProximityPrompt") then
                        pcall(function()
                            fireproximityprompt(descendant)
                        end)
                        break
                    end
                end
                
                task.wait(0.5)
                player.Character.HumanoidRootPart.CFrame = teleportCFrame
                
                return
            end
        end
    end
end

local function getMutationLabelFromModel(model)
    for _, descendant in pairs(model:GetDescendants()) do
        if descendant.Name == "MutationLabel" and descendant:IsA("TextLabel") then
            return descendant.Text
        end
    end
    return "Normal"
end

local function getUnlockedList(mutation)
    local success, result = pcall(function()
        local DataService = require(ReplicatedStorage:WaitForChild("Utilities"):WaitForChild("DataService"))
        return DataService.client:get({"index", mutation})
    end)
    
    if success and result and type(result) == "table" then
        return result
    end
    
    return {}
end

local function collectIndexEntity()
    if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then
        return
    end
    
    local entitiesFolder = workspace:FindFirstChild("EntitiesFolder")
    if not entitiesFolder then return end
    
    for _, model in pairs(entitiesFolder:GetChildren()) do
        if model:IsA("Model") then
            local modelMutation = getMutationLabelFromModel(model)
            
            if selectedMutations[modelMutation] then
                local unlockedList = getUnlockedList(modelMutation)
                local isUnlocked = table.find(unlockedList, model.Name) ~= nil
                
                if not isUnlocked then
                    local meshPart = nil
                    for _, descendant in pairs(model:GetDescendants()) do
                        if descendant.ClassName == "MeshPart" then
                            meshPart = descendant
                            break
                        end
                    end
                    
                    if meshPart then
                        player.Character.HumanoidRootPart.CFrame = meshPart.CFrame
                        task.wait(0.3)
                        
                        for _, descendant in pairs(model:GetDescendants()) do
                            if descendant:IsA("ProximityPrompt") then
                                pcall(function()
                                    fireproximityprompt(descendant)
                                end)
                                break
                            end
                        end
                        
                        task.wait(0.5)
                        player.Character.HumanoidRootPart.CFrame = teleportCFrame
                        
                        return
                    end
                end
            end
        end
    end
end

local brainrotLastFire = 0
local brainrotCooldown = 0.1
local moneyLastApply = 0
local moneyCooldown = 0.2

local myMoneyPlot = nil
local cachedMoneyParts = {}

local function isMyMoneyPlot(plot)
    local billboard = plot:FindFirstChild("HomeBillboard", true)
    if not billboard then return false end

    for _, v in pairs(billboard:GetDescendants()) do
        if v:IsA("TextLabel") then
            local text = string.lower(v.Text)
            local name = string.lower(player.Name)
            local display = string.lower(player.DisplayName)

            if string.find(text, name) or string.find(text, display) then
                return true
            end
        end
    end

    return false
end

local function findMyMoneyPlot()
    for _, plot in pairs(workspace:GetDescendants()) do
        if plot:IsA("Model") and plot.Name == "PlotTemplate" then
            if isMyMoneyPlot(plot) then
                return plot
            end
        end
    end
    return nil
end

local function refreshMoneyParts()
    cachedMoneyParts = {}

    if not myMoneyPlot or not myMoneyPlot.Parent then
        myMoneyPlot = findMyMoneyPlot()
    end

    if not myMoneyPlot then return end

    for _, v in pairs(myMoneyPlot:GetDescendants()) do
        if v:IsA("BasePart") and v.Name == "TouchPart" then
            table.insert(cachedMoneyParts, v)
        end
    end
end

task.spawn(function()
    while true do
        task.wait(1)
        refreshMoneyParts()
    end
end)

task.spawn(function()
    while true do
        if tick() - lastScan > 5 then
            lastScan = tick()
        end

        if autoMoney and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
            local currentTime = tick()
            if currentTime - moneyLastApply >= moneyCooldown then
                local char = player.Character
                local root = char:FindFirstChild("HumanoidRootPart")
                if root then
                    for _, v in pairs(cachedMoneyParts) do
                        if v and v.Parent then
                            firetouchinterest(root, v, 0)
                            firetouchinterest(root, v, 1)
                        end
                    end
                end
                moneyLastApply = currentTime
            end

            local plots = workspace:FindFirstChild("Plots")
            if plots then
                for _, model in pairs(plots:GetChildren()) do
                    if model:IsA("Model") and model.Name == "PlotTemplate" then
                        local collectAll = model:FindFirstChild("CollectAll")
                        if collectAll then
                            collectAll:Destroy()
                        end
                    end
                end
            end
        end

        task.wait(0.4)
    end
end)

        if autoUpgrade then
            pcall(function()
                RemoteBank.UpgradeBaseSign:FireServer()
            end)
        end

        if autoUpgradeBrainrots then
            local currentTime = tick()
            if currentTime - brainrotLastFire >= brainrotCooldown then
                for i = 1, 50 do
                    pcall(function()
                        UpgradeRemote:FireServer(i)
                    end)
                end
                brainrotLastFire = currentTime
            end
        end

        if autoRebirth then
            pcall(function()
                RemoteBank.Rebirth:InvokeServer()
            end)
        end
